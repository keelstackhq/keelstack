## Shared interfaces (every task codes against these)

```rust
// src/exec.rs
pub struct Cmd { pub program: String, pub args: Vec<String>, pub dir: Option<PathBuf>,
                 pub env: Vec<(String, String)>, pub stdin: Option<zeroize::Zeroizing<Vec<u8>>> }
pub struct Out { pub code: i32, pub stdout: String, pub stderr: String }  // code -1 = not found
pub trait Runner { fn run(&self, cmd: &Cmd) -> Out; }
pub struct SystemRunner;                 // std::process; stdin piped when Some; never logs stdin
pub struct FakeRunner { .. }             // #[cfg(any(test, feature="fake"))]-free: plain pub, in src/exec.rs
// FakeRunner::new().on(&["gh","api","user"], Out{..}) matches by argv prefix (longest wins),
// records every Cmd (calls()), unknown command -> Out{code:127,..}.

// src/ctx.rs
pub struct Ctx<'a> { pub runner: &'a dyn Runner, pub dir: PathBuf, pub manifest: Manifest,
                     pub pins: Pins, pub opts: Opts, pub sleep: &'a dyn Fn(Duration),
                     pub stdin: &'a dyn Fn() -> std::io::Result<Zeroizing<String>> }
pub struct Opts { pub wait_secs: u64, pub poll_secs: u64, pub token_stdin: bool, pub dry_run: bool }

// src/step.rs
pub enum Check { Done(String), Todo(String), Human(HumanStep), Failed(String) }
pub trait Step { fn id(&self) -> &'static str;
                 fn check(&self, cx: &mut Ctx) -> Result<Check, KeelError>;
                 fn apply(&self, cx: &mut Ctx) -> Result<Check, KeelError>; }  // returns re-check

// src/report.rs
pub struct StepResult { pub id: String, pub action: Action, pub detail: String }
pub enum Action { Ok, Done, WouldRun, Pending, Failed, NotReached }   // serde kebab-case
pub struct HumanStep { pub id: String, pub title: String, pub url: String, pub command: String }
pub struct Report { pub command: String, pub status: Status, pub steps: Vec<StepResult>,
                    pub human_step: Option<HumanStep>, pub exit_code: i32 }
pub enum Status { Done, Human, Failed, Planned }   // "error" only in render_json_error
pub fn render_json(&Report) -> String; pub fn render_json_error(cmd, msg, fix) -> String;
pub fn render_text(&Report) -> String;

// src/error.rs: KeelError { Usage(String), Manifest{file,message}, Env(String) } -> exit 3
```

Engine (`src/engine.rs`): `run_steps(cx, steps, mode)` with mode `Up | Plan`. Up: for each step,
check; Done → `ok`; Todo → apply → Done → `done`, Human → `pending` + stop, Failed → `failed` +
stop. Plan: check only; Todo → `would-run`, continue; Human → `pending` + stop. Remaining steps →
`not-reached`. Exit: done 0, human 2, failed 1, plan with would-run 1, plan all ok 0.
